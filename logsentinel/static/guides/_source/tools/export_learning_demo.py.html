Repository source snapshot packaged with this release.

   1 """Export a self-contained experiment with results measured by the real engine."""
   2 
   3 import argparse
   4 import json
   5 from pathlib import Path
   6 
   7 from logsentinel.detection import DEFAULTS
   8 from logsentinel.lab import SCENARIOS, replay, scenario_events
   9 
  10 
  11 def export(output):
  12     windows = [10, 60, 120, 225, 299, 300, 301, 600]
  13     scenarios = []
  14     for scenario in SCENARIOS:
  15         events = scenario_events(scenario["id"])
  16         results = {}
  17         for threshold in range(2, 9):
  18             for window in windows:
  19                 # Vary failure-only thresholds together; success threshold stays at 3.
  20                 settings = {
  21                     **DEFAULTS,
  22                     "ssh_threshold": threshold,
  23                     "windows_threshold": threshold,
  24                     "sudo_threshold": threshold,
  25                     "window_seconds": window,
  26                 }
  27                 measured = replay(events, settings)
  28                 results[f"{threshold}:{window}"] = [
  29                     {
  30                         "count": step["alerts_total"],
  31                         "eligible": step["eligible"],
  32                         "reasons": [
  33                             a["reason"] for a in step["triggered"] + step["updated"]
  34                         ],
  35                     }
  36                     for step in measured["steps"]
  37                 ]
  38         scenarios.append({**scenario, "events": events, "results": results})
  39     data = json.dumps(
  40         {"windows": windows, "scenarios": scenarios}, ensure_ascii=False
  41     ).replace("<", "\\u003c")
  42     template = (Path(__file__).parent / "learning_demo_template.html").read_text()
  43     output.write_text(template.replace("__MEASURED_RESULTS__", data))
  44     print(
  45         f"Exported {len(scenarios) * 7 * len(windows)} measured parameter combinations to {output}"
  46     )
  47 
  48 
  49 if __name__ == "__main__":
  50     parser = argparse.ArgumentParser()
  51     parser.add_argument("output", type=Path)
  52     export(parser.parse_args().output)
