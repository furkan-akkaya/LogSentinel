  31 def xml(value):
  32     root = ET.Element(
  33         "Event", xmlns="http://schemas.microsoft.com/win/2004/08/events/event"
  34     )
  35     system = ET.SubElement(root, "System")
  36     for name, data in value["System"].items():
  37         item = ET.SubElement(system, name)
  38         if isinstance(data, dict):
  39             item.attrib.update(data)
  40         else:
  41             item.text = str(data)
  42     data = ET.SubElement(root, "EventData")
  43     for name, value in value["EventData"].items():
  44         ET.SubElement(data, "Data", Name=name).text = value
  45     return ET.tostring(root, encoding="unicode")

  48 @pytest.mark.parametrize("format", ["json", "xml"])
  49 @pytest.mark.parametrize(
  50     "change",
  51     [
  52         {"Channel": "Application"},
  53         {"Provider": {"Name": "Other-Provider"}},
  54         {"Provider": {"Guid": "{00000000-0000-0000-0000-000000000000}"}},
  55         {"Provider": {"Name": SECURITY_PROVIDER, "Guid": "not-a-guid"}},
  56         {"Provider": {"Name": "Other", "Guid": GUID}},

  61 def test_conflicting_source_rejected(format, change):
  62     value = record()
  63     value["System"].update(change)
  64     result = parse(xml(value) if format == "xml" else json.dumps(value), format)
  65     assert result.events == [] and result.rejected == 1
  66     assert result.errors[0]["reason"].startswith("Windows source rejected:")

 106 @pytest.mark.parametrize("missing", ["Channel", "Provider", "both"])
 107 @pytest.mark.parametrize("format", ["json", "xml"])
 108 def test_absent_source_fields_keep_compatibility(missing, format):
 109     value = record()
 110     value["System"]["Provider"] = {"Name": SECURITY_PROVIDER, "Guid": GUID}
 111     if missing in ("Channel", "both"):
 112         value["System"].pop("Channel")
 113     if missing in ("Provider", "both"):
 114         value["System"].pop("Provider")
 115     result = parse(xml(value) if format == "xml" else json.dumps(value), format)
 116     assert len(result.events) == 1 and result.rejected == 0

 136 def test_wrong_source_cannot_generate_alert_and_partial_import_survives(client):
 137     bad = []
 138     for i in range(5):
 139         obj = record()
 140         obj["System"].update(
 141             Channel="Application", Provider={"Name": "Other"}, EventRecordID=str(i)
 142         )
 143         bad.append(obj)
 144     result = client.post(
 145         "/api/import", files={"file": ("bad.json", json.dumps(bad))}
 146     ).json()
 147     assert (
 148         result["rejected"] == 5
 149         and result["inserted"] == 0
 150         and result["alerts_total"] == 0
 151     )
 152     good = record()
 153     good["System"]["Provider"] = {"Name": SECURITY_PROVIDER, "Guid": GUID}
 154     result = client.post(
 155         "/api/import", files={"file": ("mixed.json", json.dumps([bad[0], good]))}
 156     ).json()
 157     assert result["inserted"] == 1 and result["rejected"] == 1
