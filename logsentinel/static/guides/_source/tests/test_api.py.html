 215 def test_alert_explanation_uses_real_counts_context_and_parameters(client):
 216     client.post("/api/demo")
 217     items = client.get("/api/alerts").json()["items"]
 218     for item in items:
 219         a = client.get("/api/alerts/" + item["id"]).json()
 220         info = a["explanation"]
 221         assert info["evidence_count"] == len(a["events"])
 222         assert info["failure_count"] + info["success_count"] == len(a["events"])
 223         assert info["parameters"] == a["parameters"]
 224         assert client.get(info["guide_url"]).status_code == 200
 225         if item["rule"].startswith("windows"):
 226             assert (
 227                 info["identity"]["domain"] == "lab"
 228                 and info["identity"]["logon_type"] == "10"
 229             )
 230     client.put(
 231         "/api/settings", json={**DEFAULTS, "window_seconds": 180, "ssh_threshold": 6}
 232     )
 233     a = next(
 234         a
 235         for a in client.get("/api/alerts").json()["items"]
 236         if a["rule"] == "ssh_brute_force"
 237     )
 238     info = client.get("/api/alerts/" + a["id"]).json()["explanation"]
 239     assert (
 240         info["parameters"] == {"threshold": 6, "window_seconds": 180}
 241         and info["failure_count"] == 7
 242     )

 245 def test_legacy_alert_explanation_without_rewriting_stored_body(client):
 246     from logsentinel.storage import connection
 247 
 248     client.post("/api/demo")
 249     with connection(client.app.state.db_path) as db:
 250         row = db.execute("select id,body from alerts limit 1").fetchone()
 251         body = json.loads(row["body"])
 252         body.pop("parameters")
 253         encoded = json.dumps(body)
 254         db.execute("update alerts set body=? where id=?", (encoded, row["id"]))
 255     a = client.get("/api/alerts/" + row["id"]).json()
 256     assert a["explanation"]["parameters"]["window_seconds"] == 300
 257     with connection(client.app.state.db_path) as db:
 258         assert (
 259             db.execute("select body from alerts where id=?", (row["id"],)).fetchone()[0]
 260             == encoded
 261         )
