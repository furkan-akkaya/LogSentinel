  30 def test_scenarios_are_real_engine_results_at_every_step(client, scenario, count):
  31     response = run(client, scenario)
  32     assert response.status_code == 200
  33     result = response.json()
  34     assert len(result["alerts"]) == count
  35     assert result["scope"]["kind"] == "synthetic_scenario"
  36     for i, step in enumerate(result["steps"]):
  37         expected = detect(result["events"][: i + 1], DEFAULTS)
  38         assert step["alerts_total"] == len(expected)
  39         assert all(a in expected for a in step["triggered"] + step["updated"])

  97 def test_replay_bounded_and_missing_alert(client):

 110 def test_notebook_survives_removed_alert_and_restart(tmp_path):
 111     path = tmp_path / "notes.db"
 112     with TestClient(create_app(path), base_url="http://localhost") as c:
 113         c.post("/api/demo")
 114         a = c.get("/api/alerts").json()["items"][0]
 115         note = {
 116             "assessment": "undetermined",
 117             "observation": "Four sudo failures",
 118             "hypothesis": "A typo is possible",
 119             "next_step": "Ask the account owner",
 120         }
 121         saved = c.put(f"/api/alerts/{a['id']}/notebook", json=note).json()
 122         assert saved["note"] == note
 123         assert saved["alert"]["evidence"] == [e["id"] for e in saved["events"]]
 124         assert saved["evidence_omitted"] == 0
 125         assert c.get(f"/api/alerts/{a['id']}").json()["notebook"] == saved
 126         c.put("/api/settings", json={**DEFAULTS, "sudo_threshold": 100})
 127         assert c.get(f"/api/alerts/{a['id']}").status_code == 404
 128         assert c.get("/api/notebooks").json()["items"][0]["active"] is False
 129     with TestClient(create_app(path), base_url="http://localhost") as c:
 130         assert c.get(f"/api/notebooks/{a['id']}").json() == saved
 131         c.put("/api/settings", json=DEFAULTS)
 132         assert c.get("/api/notebooks").json()["items"][0]["active"] is True
 133         updated = c.put(
 134             f"/api/alerts/{a['id']}/notebook",
 135             json={**note, "assessment": "likely_benign"},
 136         ).json()
 137         assert updated["note"]["assessment"] == "likely_benign"
 138         assert len(c.get("/api/notebooks").json()["items"]) == 1
 139         assert saved["note"]["assessment"] == "undetermined"

 208 def test_large_alert_has_explicit_snapshot_limit_and_replay_refusal(client):
 209     from datetime import datetime, timedelta, timezone
 210 
 211     records = []
 212     for i in range(250):
 213         stamp = (
 214             datetime(2026, 10, 26, tzinfo=timezone.utc) + timedelta(seconds=i)
 215         ).isoformat()
 216         records.append(
 217             f"{stamp} dense sshd[100]: Failed password for alice from 192.0.2.9 port {40000 + i} ssh2"
 218         )
 219     result = client.post(
 220         "/api/import", files={"file": ("dense.log", "\n".join(records))}
 221     )
 222     assert result.json()["inserted"] == 250
 223     alert = client.get("/api/alerts").json()["items"][0]
 224     assert len(alert["evidence"]) == 250
 225     assert (
 226         client.post(
 227             "/api/lab/replay", json={"alert_id": alert["id"], "settings": DEFAULTS}
 228         ).status_code
 229         == 422
 230     )
 231     saved = client.put(
 232         f"/api/alerts/{alert['id']}/notebook", json={"observation": "Dense sequence"}
 233     ).json()
 234     assert len(saved["events"]) == 200
 235     assert saved["evidence_omitted"] == 50
 236     assert saved["snapshot_policy"] == "first_and_final"
 237     assert saved["events"][-1]["id"] == alert["evidence"][-1]
 238     assert [e["id"] for e in saved["events"][:-1]] == alert["evidence"][:199]
 239     assert saved["alert"]["evidence"] == alert["evidence"]
