   6 def event(number, kind="ssh_failure", second=None, **changes):
   7     dt = datetime(2026, 10, 26, tzinfo=timezone.utc) + timedelta(
   8         seconds=number if second is None else second
   9     )
  10     value = {
  11         "id": str(number),
  12         "timestamp": dt.isoformat(timespec="microseconds").replace("+00:00", "Z"),
  13         "source": "windows" if kind.startswith("windows") else "linux",
  14         "event_type": kind,
  15         "host": "server",
  16         "user": "alice",
  17         "source_ip": "192.0.2.1",
  18         "metadata": {"domain": "LAB", "logon_type": "10", "tty": "pts/1"},
  19     }
  20     value.update(changes)
  21     return value
