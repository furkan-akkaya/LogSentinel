  59 def scenario_events(identifier):
  60     """Generate documented synthetic records through the actual parsers."""
  61     if identifier not in {s["id"] for s in SCENARIOS}:
  62         raise ValueError("Unknown lab scenario")
  63     if identifier == "windows_context":
  64         import json
  65 
  66         records = [
  67             {
  68                 "System": {
  69                     "EventID": 4625 if i < 4 else 4624,
  70                     "Computer": "lab-windows",
  71                     "TimeCreated": f"2026-10-26T12:00:0{i}Z",
  72                 },
  73                 "EventData": {
  74                     "TargetUserName": "analyst",
  75                     "TargetDomainName": "LAB",
  76                     "IpAddress": "192.0.2.20",
  77                     "LogonType": "10" if i < 4 else "3",
  78                 },
  79             }
  80             for i in range(5)
  81         ]
  82         parsed = parse(json.dumps(records), "json")
  83     else:
  84         offsets = (
  85             [0, 75, 150, 225, 300] if identifier == "window_edge" else list(range(6))
  86         )
  87         records = []
  88         for i, offset in enumerate(offsets):
  89             stamp = datetime(2026, 10, 26, 12, tzinfo=timezone.utc) + timedelta(
  90                 seconds=offset
  91             )
  92             outcome = (
  93                 "Accepted" if identifier == "ssh_recovery" and i == 5 else "Failed"
  94             )
  95             method = "publickey" if identifier == "public_keys" else "password"
  96             user = f"user{i}" if identifier == "split_identity" else "analyst"
  97             records.append(
  98                 f"{stamp.isoformat()} lab-ssh sshd[100]: {outcome} {method} for {user} from 192.0.2.10 port {40000 + i} ssh2"
  99             )
 100         parsed = parse("\n".join(records), "linux")
 101     assert not parsed.rejected and not parsed.ignored
 102     return [
 103         {**asdict(e), "id": e.id, "filename": "isolated synthetic lab"}
 104         for e in parsed.events
 105     ]

 108 def replay(events, parameters):
 109     """Replay sorted prefixes. Never write events, settings, alerts, or statuses."""
 110     if len(events) > MAX_REPLAY_EVENTS:
 111         raise ValueError(
 112             f"Replay supports at most {MAX_REPLAY_EVENTS} evidence records; this alert is too large"
 113         )
 114     ordered = sorted(events, key=lambda e: (e["timestamp"], e["id"]))
 115     steps, previous = [], {}
 116     for index, event in enumerate(ordered):
 117         alerts = detect(ordered[: index + 1], parameters)
 118         current = {a["id"]: a for a in alerts}
 119         steps.append(
 120             {
 121                 "event_id": event["id"],
 122                 "eligible": correlation_key(event) is not None,
 123                 "identity": list(correlation_key(event) or []),
 124                 "alerts_total": len(alerts),
 125                 "triggered": [a for key, a in current.items() if key not in previous],
 126                 "updated": [
 127                     a
 128                     for key, a in current.items()
 129                     if key in previous and a["evidence"] != previous[key]["evidence"]
 130                 ],
 131             }
 132         )
 133         previous = current
 134     return {
 135         "events": ordered,
 136         "steps": steps,
 137         "alerts": list(previous.values()),
 138         "settings": parameters,
 139     }
