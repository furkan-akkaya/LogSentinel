  26 class LocalBoundary:
  27     """Bound request bodies before multipart parsing; reject cross-origin writes."""
  28 
  29     def __init__(self, app):
  30         self.app = app
  31 
  32     async def __call__(self, scope, receive, send):
  33         if scope["type"] != "http":
  34             return await self.app(scope, receive, send)
  35         headers = dict(scope["headers"])
  36         mutating = scope["method"] in ("POST", "PUT", "PATCH", "DELETE")
  37         origin = headers.get(b"origin")
  38         host = headers.get(b"host", b"").decode()
  39         if (
  40             mutating
  41             and origin
  42             and origin.decode() != f"{scope.get('scheme', 'http')}://{host}"
  43         ):
  44             return await JSONResponse(
  45                 {"detail": "Cross-origin writes are disabled"}, 403
  46             )(scope, receive, send)
  47         if mutating:
  48             chunks, size = [], 0
  49             while True:
  50                 message = await receive()
  51                 if message["type"] == "http.disconnect":
  52                     return
  53                 size += len(message.get("body", b""))
  54                 if size > MAX_UPLOAD + 65_536:
  55                     return await JSONResponse(
  56                         {"detail": "Request exceeds 5 MiB upload limit"}, 413
  57                     )(scope, receive, send)
  58                 chunks.append(message.get("body", b""))
  59                 if not message.get("more_body"):
  60                     break
  61             delivered = False
  62 
  63             async def replay():
  64                 nonlocal delivered
  65                 if not delivered:
  66                     delivered = True
  67                     return {
  68                         "type": "http.request",
  69                         "body": b"".join(chunks),
  70                         "more_body": False,
  71                     }
  72                 return await receive()
  73 
  74             await self.app(scope, replay, send)
  75         else:
  76             await self.app(scope, receive, send)

 146     @app.post("/api/import")
 147     def import_logs(
 148         file: UploadFile = File(...),
 149         format: Literal["auto", "linux", "json", "xml"] = Form("auto"),
 150         year: int = Form(datetime.now(timezone.utc).year),
 151         timezone_offset: str = Form("+00:00"),
 152     ):
 153         content = file.file.read(MAX_UPLOAD + 1)
 154         if len(content) > MAX_UPLOAD:
 155             raise HTTPException(413, "File exceeds 5 MiB")
 156         try:
 157             parsed = parse(content.decode("utf-8-sig"), format, year, timezone_offset)
 158             with connection(path) as db:
 159                 result = ingest(
 160                     db, parsed.events, Path(file.filename or "upload").name[:200]
 161                 )
 162         except UnicodeDecodeError:
 163             raise HTTPException(
 164                 400,
 165                 "Use UTF-8 Linux text, Windows XML or JSON; binary EVTX is unsupported",
 166             )
 167         except ValueError as exc:
 168             raise HTTPException(400, str(exc))
 169         return {
 170             **result,
 171             "rejected": parsed.rejected,
 172             "ignored": parsed.ignored,
 173             "errors": parsed.errors,
 174             "time_context": {"year": year, "timezone_offset": timezone_offset},
 175         }

 216     @app.get("/api/events")
 217     def events(
 218         q: str = Query("", max_length=200),
 219         source: Literal["linux", "windows"] | None = None,
 220         event_type: str = Query("", max_length=80),
 221         source_ip: str = Query("", max_length=64),
 222         user: str = Query("", max_length=200),
 223         host: str = Query("", max_length=200),
 224         start: str | None = None,
 225         end: str | None = None,
 226         limit: int = Query(50, ge=1, le=200),
 227         offset: int = Query(0, ge=0),
 228     ):
 229         clauses, args = [], []
 230         for field, value in [
 231             ("source", source),
 232             ("event_type", event_type),
 233             ("source_ip", source_ip),
 234             ("user", user),
 235             ("host", host),
 236         ]:
 237             if value:
 238                 clauses.append(f"{field}=?")
 239                 args.append(value)
 240         try:
 241             if start:
 242                 clauses.append("timestamp>=?")
 243                 args.append(timestamp(start))
 244             if end:
 245                 clauses.append("timestamp<=?")
 246                 args.append(timestamp(end))
 247             if start and end and timestamp(start) > timestamp(end):
 248                 raise ValueError("Start must precede end")
 249         except (ValueError, TypeError):
 250             raise HTTPException(422, "Invalid date range; use ISO 8601")
 251         if q:
 252             clauses.append(
 253                 "(instr(lower(raw), lower(?))>0 OR instr(lower(host), lower(?))>0 OR instr(lower(coalesce(user,'')), lower(?))>0)"
 254             )
 255             args.extend([q, q, q])
 256         where = " WHERE " + " AND ".join(clauses) if clauses else ""
 257         with connection(path) as db:
 258             total = db.execute("SELECT count(*) FROM events" + where, args).fetchone()[
 259                 0
 260             ]
 261             rows = db.execute(
 262                 "SELECT * FROM events"
 263                 + where
 264                 + " ORDER BY timestamp DESC,id LIMIT ? OFFSET ?",
 265                 args + [limit, offset],
 266             )
 267             items = [event_dict(row) for row in rows]
 268         return {"total": total, "items": items, "limit": limit, "offset": offset}

 280     @app.get("/api/alerts")
 281     def alerts(
 282         severity: Literal["critical", "high", "medium"] | None = None,
 283         status: Literal["new", "investigating", "resolved"] | None = None,
 284         limit: int = Query(50, ge=1, le=200),
 285         offset: int = Query(0, ge=0),
 286     ):
 287         with connection(path) as db:
 288             items = [
 289                 {**json.loads(row["body"]), "status": row["status"]}
 290                 for row in db.execute("SELECT * FROM alerts")
 291             ]
 292         items = [
 293             a
 294             for a in items
 295             if (not severity or a["severity"] == severity)
 296             and (not status or a["status"] == status)
 297         ]
 298         items.sort(key=lambda a: (a["last_seen"], a["id"]), reverse=True)
 299         return {"total": len(items), "items": items[offset : offset + limit]}

 301     @app.get("/api/alerts/{identifier}")
 302     def alert_detail(identifier: str):
 303         with connection(path) as db:
 304             # One read snapshot keeps legacy alert parameters consistent with
 305             # their evidence/config even during an overlapping settings update.
 306             db.execute("BEGIN")
 307             row = db.execute(
 308                 "SELECT * FROM alerts WHERE id=?", (identifier,)
 309             ).fetchone()
 310             if row is None:
 311                 raise HTTPException(404, "Alert not found")
 312             item = {**json.loads(row["body"]), "status": row["status"]}
 313             # Separate indexed queries avoid SQLite variable limits for long episodes.
 314             item["events"] = [
 315                 event_dict(
 316                     db.execute("SELECT * FROM events WHERE id=?", (eid,)).fetchone()
 317                 )
 318                 for eid in item["evidence"]
 319             ]
 320             item["explanation"] = explain_alert(item, item["events"], settings(db))
 321             return item

 333     @app.get("/api/stats")
 334     def stats():
 335         with connection(path) as db:
 336             count = db.execute("SELECT count(*) FROM events").fetchone()[0]
 337             hosts = db.execute("SELECT count(DISTINCT host) FROM events").fetchone()[0]
 338             latest = db.execute("SELECT max(timestamp) FROM events").fetchone()[0]
 339             sources = dict(
 340                 db.execute("SELECT source,count(*) FROM events GROUP BY source")
 341             )
 342             event_timeline = dict(
 343                 db.execute(
 344                     "SELECT substr(timestamp,1,13),count(*) FROM events GROUP BY substr(timestamp,1,13) ORDER BY 1"
 345                 )
 346             )
 347             alerts = [
 348                 {**json.loads(r["body"]), "status": r["status"]}
 349                 for r in db.execute("SELECT * FROM alerts")
 350             ]
 351             gaps = sum(
 352                 correlation_key(event_dict(r)) is None
 353                 for r in db.execute("SELECT * FROM events")
 354             )
 355         severity = {
 356             key: sum(a["severity"] == key for a in alerts)
 357             for key in ("critical", "high", "medium")
 358         }
 359         alert_timeline = {}
 360         for a in alerts:
 361             hour = a["last_seen"][:13]
 362             alert_timeline[hour] = alert_timeline.get(hour, 0) + 1
 363         return {
 364             "events": count,
 365             "alerts": len(alerts),
 366             "hosts": hosts,
 367             "latest_event": latest,
 368             "sources": sources,
 369             "severity": severity,
 370             "uncorrelated_events": gaps,
 371             "open_alerts": sum(a["status"] != "resolved" for a in alerts),
 372             "timeline": [
 373                 {
 374                     "hour": h + ":00:00Z",
 375                     "events": event_timeline.get(h, 0),
 376                     "alerts": alert_timeline.get(h, 0),
 377                 }
 378                 for h in sorted(set(event_timeline) | set(alert_timeline))
 379             ],
 380             "recent_alerts": sorted(alerts, key=lambda a: a["last_seen"], reverse=True)[
 381                 :5
 382             ],
 383         }
