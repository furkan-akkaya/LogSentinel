  30 class LocalBoundary:
  31     """Bound request bodies before multipart parsing; reject cross-origin writes."""

  33     def __init__(self, app):
  34         self.app = app

  36     async def __call__(self, scope, receive, send):
  37         if scope["type"] != "http":
  38             return await self.app(scope, receive, send)
  39         headers = dict(scope["headers"])
  40         mutating = scope["method"] in ("POST", "PUT", "PATCH", "DELETE")
  41         origin = headers.get(b"origin")
  42         host = headers.get(b"host", b"").decode()
  43         if (
  44             mutating
  45             and origin
  46             and origin.decode() != f"{scope.get('scheme', 'http')}://{host}"
  47         ):
  48             return await JSONResponse(
  49                 {"detail": "Cross-origin writes are disabled"}, 403
  50             )(scope, receive, send)
  51         if mutating:
  52             chunks, size = [], 0
  53             while True:
  54                 message = await receive()
  55                 if message["type"] == "http.disconnect":
  56                     return
  57                 size += len(message.get("body", b""))
  58                 if size > MAX_UPLOAD + 65_536:
  59                     return await JSONResponse(
  60                         {"detail": "Request exceeds 5 MiB upload limit"}, 413
  61                     )(scope, receive, send)
  62                 chunks.append(message.get("body", b""))
  63                 if not message.get("more_body"):
  64                     break
  65             delivered = False
  66 
  67             async def replay():
  68                 nonlocal delivered
  69                 if not delivered:
  70                     delivered = True
  71                     return {
  72                         "type": "http.request",
  73                         "body": b"".join(chunks),
  74                         "more_body": False,
  75                     }
  76                 return await receive()
  77 
  78             await self.app(scope, replay, send)
  79         else:
  80             await self.app(scope, receive, send)

 165     @app.post("/api/import")
 166     def import_logs(
 167         file: UploadFile = File(...),
 168         format: Literal["auto", "linux", "json", "xml"] = Form("auto"),
 169         year: int = Form(datetime.now(timezone.utc).year),
 170         timezone_offset: str = Form("+00:00"),
 171     ):
 172         content = file.file.read(MAX_UPLOAD + 1)
 173         if len(content) > MAX_UPLOAD:
 174             raise HTTPException(413, "File exceeds 5 MiB")
 175         try:
 176             parsed = parse(content.decode("utf-8-sig"), format, year, timezone_offset)
 177             with connection(path) as db:
 178                 result = ingest(
 179                     db, parsed.events, Path(file.filename or "upload").name[:200]
 180                 )
 181         except UnicodeDecodeError:
 182             raise HTTPException(
 183                 400,
 184                 "Use UTF-8 Linux text, Windows XML or JSON; binary EVTX is unsupported",
 185             )
 186         except ValueError as exc:
 187             raise HTTPException(400, str(exc))
 188         return {
 189             **result,
 190             "rejected": parsed.rejected,
 191             "ignored": parsed.ignored,
 192             "errors": parsed.errors,
 193             "time_context": {"year": year, "timezone_offset": timezone_offset},
 194         }

 235     @app.get("/api/events")
 236     def events(
 237         q: str = Query("", max_length=200),
 238         source: Literal["linux", "windows"] | None = None,
 239         event_type: str = Query("", max_length=80),
 240         source_ip: str = Query("", max_length=64),
 241         user: str = Query("", max_length=200),
 242         host: str = Query("", max_length=200),
 243         start: str | None = None,
 244         end: str | None = None,
 245         limit: int = Query(50, ge=1, le=200),
 246         offset: int = Query(0, ge=0),
 247     ):
 248         clauses, args = [], []
 249         for field, value in [
 250             ("source", source),
 251             ("event_type", event_type),
 252             ("source_ip", source_ip),
 253             ("user", user),
 254             ("host", host),
 255         ]:
 256             if value:
 257                 clauses.append(f"{field}=?")
 258                 args.append(value)
 259         try:
 260             if start:
 261                 clauses.append("timestamp>=?")
 262                 args.append(timestamp(start))
 263             if end:
 264                 clauses.append("timestamp<=?")
 265                 args.append(timestamp(end))
 266             if start and end and timestamp(start) > timestamp(end):
 267                 raise ValueError("Start must precede end")
 268         except (ValueError, TypeError):
 269             raise HTTPException(422, "Invalid date range; use ISO 8601")
 270         if q:
 271             clauses.append(
 272                 "(instr(lower(raw), lower(?))>0 OR instr(lower(host), lower(?))>0 OR instr(lower(coalesce(user,'')), lower(?))>0)"
 273             )
 274             args.extend([q, q, q])
 275         where = " WHERE " + " AND ".join(clauses) if clauses else ""
 276         with connection(path) as db:
 277             total = db.execute("SELECT count(*) FROM events" + where, args).fetchone()[
 278                 0
 279             ]
 280             rows = db.execute(
 281                 "SELECT * FROM events"
 282                 + where
 283                 + " ORDER BY timestamp DESC,id LIMIT ? OFFSET ?",
 284                 args + [limit, offset],
 285             )
 286             items = [event_dict(row) for row in rows]
 287         return {"total": total, "items": items, "limit": limit, "offset": offset}

 299     @app.get("/api/alerts")
 300     def alerts(
 301         severity: Literal["critical", "high", "medium"] | None = None,
 302         status: Literal["new", "investigating", "resolved"] | None = None,
 303         limit: int = Query(50, ge=1, le=200),
 304         offset: int = Query(0, ge=0),
 305     ):
 306         with connection(path) as db:
 307             items = [
 308                 {**json.loads(row["body"]), "status": row["status"]}
 309                 for row in db.execute("SELECT * FROM alerts")
 310             ]
 311         items = [
 312             a
 313             for a in items
 314             if (not severity or a["severity"] == severity)
 315             and (not status or a["status"] == status)
 316         ]
 317         items.sort(key=lambda a: (a["last_seen"], a["id"]), reverse=True)
 318         return {"total": len(items), "items": items[offset : offset + limit]}

 320     @app.get("/api/alerts/{identifier}")
 321     def alert_detail(identifier: str):
 322         with connection(path) as db:
 323             # One read snapshot keeps legacy alert parameters consistent with
 324             # their evidence/config even during an overlapping settings update.
 325             db.execute("BEGIN")
 326             row = db.execute(
 327                 "SELECT * FROM alerts WHERE id=?", (identifier,)
 328             ).fetchone()
 329             if row is None:
 330                 raise HTTPException(404, "Alert not found")
 331             item = {**json.loads(row["body"]), "status": row["status"]}
 332             # Separate indexed queries avoid SQLite variable limits for long episodes.
 333             item["events"] = [
 334                 event_dict(
 335                     db.execute("SELECT * FROM events WHERE id=?", (eid,)).fetchone()
 336                 )
 337                 for eid in item["evidence"]
 338             ]
 339             item["explanation"] = explain_alert(item, item["events"], settings(db))
 340             item["dataset"] = dataset_summary(item["events"])
 341             note = db.execute(
 342                 "SELECT body FROM notebooks WHERE alert_id=?", (identifier,)
 343             ).fetchone()
 344             item["notebook"] = json.loads(note["body"]) if note else None
 345             return item

 351     @app.post("/api/lab/replay")
 352     def lab_replay(value: ReplayRequest):
 353         if bool(value.scenario) == bool(value.alert_id):
 354             raise HTTPException(422, "Choose exactly one scenario or alert_id")
 355         if value.alert_id:
 356             with connection(path) as db:
 357                 db.execute("BEGIN")
 358                 row = db.execute(
 359                     "SELECT body FROM alerts WHERE id=?", (value.alert_id,)
 360                 ).fetchone()
 361                 if row is None:
 362                     raise HTTPException(404, "Alert not found")
 363                 alert = json.loads(row["body"])
 364                 baseline = settings(db)
 365                 if len(alert["evidence"]) > MAX_REPLAY_EVENTS:
 366                     raise HTTPException(
 367                         422,
 368                         f"Replay supports at most {MAX_REPLAY_EVENTS} evidence records; this alert is too large",
 369                     )
 370                 events = [
 371                     event_dict(
 372                         db.execute("SELECT * FROM events WHERE id=?", (eid,)).fetchone()
 373                     )
 374                     for eid in alert["evidence"]
 375                 ]
 376             scope = {
 377                 "kind": "alert_evidence",
 378                 "alert_id": value.alert_id,
 379                 "title": alert["title"],
 380                 "dataset": dataset_summary(events),
 381                 "limits": "Only this alert's saved evidence is replayed. Other stored events are excluded. This is not a coverage or false-positive measurement.",
 382             }
 383         else:
 384             try:
 385                 events = scenario_events(value.scenario)
 386             except ValueError as exc:
 387                 raise HTTPException(422, str(exc))
 388             scope = {
 389                 "kind": "synthetic_scenario",
 390                 "scenario": value.scenario,
 391                 "title": next(
 392                     s["title"] for s in SCENARIOS if s["id"] == value.scenario
 393                 ),
 394                 "limits": "Synthetic records in an isolated experiment. They are never imported into your workspace.",
 395             }
 396             with connection(path) as db:
 397                 baseline = settings(db)
 398         return {
 399             **replay(events, value.settings.model_dump()),
 400             "scope": scope,
 401             "baseline": {
 402                 "settings": baseline,
 403                 "alerts_total": len(detect(events, baseline)),
 404             },
 405         }

 407     @app.put("/api/alerts/{identifier}/notebook")
 408     def save_notebook(identifier: str, value: AnalystNote):
 409         with connection(path) as db:
 410             db.execute("BEGIN IMMEDIATE")
 411             row = db.execute(
 412                 "SELECT * FROM alerts WHERE id=?", (identifier,)
 413             ).fetchone()
 414             if row is None:
 415                 raise HTTPException(404, "Alert not found")
 416             alert = {**json.loads(row["body"]), "status": row["status"]}
 417             evidence_ids = alert["evidence"]
 418             snapshot_ids = (
 419                 evidence_ids
 420                 if len(evidence_ids) <= MAX_NOTE_EVIDENCE
 421                 else evidence_ids[: MAX_NOTE_EVIDENCE - 1] + evidence_ids[-1:]
 422             )
 423             events = [
 424                 event_dict(
 425                     db.execute("SELECT * FROM events WHERE id=?", (eid,)).fetchone()
 426                 )
 427                 for eid in snapshot_ids
 428             ]
 429             item = {
 430                 "alert_id": identifier,
 431                 "saved_at": datetime.now(timezone.utc).isoformat(),
 432                 "note": value.model_dump(),
 433                 "alert": alert,
 434                 "events": events,
 435                 "snapshot_policy": "all"
 436                 if len(evidence_ids) <= MAX_NOTE_EVIDENCE
 437                 else "first_and_final",
 438                 "evidence_omitted": max(0, len(alert["evidence"]) - len(events)),
 439                 "scope": "Manual analyst notes and a bounded evidence snapshot. Not a verified verdict or an audit history.",
 440             }
 441             db.execute(
 442                 "INSERT INTO notebooks VALUES (?,?) ON CONFLICT(alert_id) DO UPDATE SET body=excluded.body",
 443                 (identifier, json.dumps(item)),
 444             )
 445         return item

 486     @app.get("/api/stats")
 487     def stats():
 488         with connection(path) as db:
 489             db.execute("BEGIN")
 490             count = db.execute("SELECT count(*) FROM events").fetchone()[0]
 491             hosts = db.execute("SELECT count(DISTINCT host) FROM events").fetchone()[0]
 492             latest = db.execute("SELECT max(timestamp) FROM events").fetchone()[0]
 493             sources = dict(
 494                 db.execute("SELECT source,count(*) FROM events GROUP BY source")
 495             )
 496             event_timeline = dict(
 497                 db.execute(
 498                     "SELECT substr(timestamp,1,13),count(*) FROM events GROUP BY substr(timestamp,1,13) ORDER BY 1"
 499                 )
 500             )
 501             alerts = [
 502                 {**json.loads(r["body"]), "status": r["status"]}
 503                 for r in db.execute("SELECT * FROM alerts")
 504             ]
 505             gaps = sum(
 506                 correlation_key(event_dict(r)) is None
 507                 for r in db.execute("SELECT * FROM events")
 508             )
 509             dataset = dataset_summary(
 510                 [dict(r) for r in db.execute("SELECT id FROM events")]
 511             )
 512         severity = {
 513             key: sum(a["severity"] == key for a in alerts)
 514             for key in ("critical", "high", "medium")
 515         }
 516         alert_timeline = {}
 517         for a in alerts:
 518             hour = a["last_seen"][:13]
 519             alert_timeline[hour] = alert_timeline.get(hour, 0) + 1
 520         return {
 521             "events": count,
 522             "dataset": dataset,
 523             "alerts": len(alerts),
 524             "hosts": hosts,
 525             "latest_event": latest,
 526             "sources": sources,
 527             "severity": severity,
 528             "uncorrelated_events": gaps,
 529             "open_alerts": sum(a["status"] != "resolved" for a in alerts),
 530             "timeline": [
 531                 {
 532                     "hour": h + ":00:00Z",
 533                     "events": event_timeline.get(h, 0),
 534                     "alerts": alert_timeline.get(h, 0),
 535                 }
 536                 for h in sorted(set(event_timeline) | set(alert_timeline))
 537             ],
 538             "recent_alerts": sorted(alerts, key=lambda a: a["last_seen"], reverse=True)[
 539                 :5
 540             ],
 541         }
