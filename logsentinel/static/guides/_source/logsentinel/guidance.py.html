  54 def explain_alert(alert, events, current_settings):
  55     key = correlation_key(events[-1])
  56     fields = (
  57         ("host", "user", "terminal")
  58         if alert["rule"] == "sudo_failures"
  59         else (
  60             ("host", "source_ip", "user", "domain", "logon_type")
  61             if events[-1]["source"] == "windows"
  62             else ("host", "source_ip", "user")
  63         )
  64     )
  65     setting_key = THRESHOLD_SETTINGS[alert["rule"]]
  66     # v1.0 alerts lack parameter snapshots. Config and alerts were updated in
  67     # one transaction, so that version's current config describes those alerts.
  68     parameters = alert.get("parameters") or {
  69         "threshold": current_settings[setting_key],
  70         "window_seconds": current_settings["window_seconds"],
  71     }
  72     return {
  73         **GUIDANCE[alert["rule"]],
  74         "failure_count": sum(e["event_type"].endswith("failure") for e in events),
  75         "success_count": sum(e["event_type"].endswith("success") for e in events),
  76         "evidence_count": len(events),
  77         "identity": dict(zip(fields, key or [])),
  78         "parameters": parameters,
  79         "threshold_setting": setting_key,
  80         "guide_url": f"/static/guides/docs/detections/{alert['rule']}.html",
  81     }
