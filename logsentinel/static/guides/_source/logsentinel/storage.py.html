   1 """SQLite persistence; imports and detection updates commit atomically."""
   2 
   3 from contextlib import contextmanager
   4 import json
   5 import os
   6 from pathlib import Path
   7 import sqlite3
   8 
   9 from .detection import DEFAULTS, detect
  10 from .provenance import dataset_kind
  11 
  12 MAX_EVENTS = 50_000
  13 
  14 
  15 @contextmanager
  16 def connection(path):
  17     db = sqlite3.connect(path, timeout=30)
  18     db.row_factory = sqlite3.Row
  19     db.execute("PRAGMA foreign_keys=ON")
  20     try:
  21         yield db
  22         db.commit()
  23     except Exception:
  24         db.rollback()
  25         raise
  26     finally:
  27         db.close()
  28 
  29 
  30 def initialize(path):
  31     Path(path).parent.mkdir(parents=True, exist_ok=True, mode=0o700)
  32     with connection(path) as db:
  33         if os.name == "posix":
  34             os.chmod(path, 0o600)
  35         db.execute("PRAGMA journal_mode=WAL")
  36         db.executescript("""
  37         CREATE TABLE IF NOT EXISTS events (
  38           id TEXT PRIMARY KEY, timestamp TEXT NOT NULL, source TEXT NOT NULL,
  39           event_type TEXT NOT NULL, host TEXT NOT NULL, user TEXT, source_ip TEXT,
  40           raw TEXT NOT NULL, metadata TEXT NOT NULL, filename TEXT NOT NULL
  41         );
  42         CREATE INDEX IF NOT EXISTS events_time ON events(timestamp);
  43         CREATE INDEX IF NOT EXISTS events_type ON events(event_type);
  44         CREATE TABLE IF NOT EXISTS alerts (
  45           id TEXT PRIMARY KEY, body TEXT NOT NULL, status TEXT NOT NULL DEFAULT 'new'
  46         );
  47         CREATE TABLE IF NOT EXISTS config (id INTEGER PRIMARY KEY CHECK(id=1), body TEXT NOT NULL);
  48         CREATE TABLE IF NOT EXISTS notebooks (
  49           alert_id TEXT PRIMARY KEY, body TEXT NOT NULL
  50         );
  51         """)
  52         db.execute(
  53             "INSERT OR IGNORE INTO config VALUES (1, ?)", (json.dumps(DEFAULTS),)
  54         )
  55 
  56 
  57 def event_dict(row):
  58     value = dict(row)
  59     value["metadata"] = json.loads(value["metadata"])
  60     value["dataset_kind"] = dataset_kind(value["id"])
  61     return value
  62 
  63 
  64 def settings(db):
  65     return json.loads(db.execute("SELECT body FROM config WHERE id=1").fetchone()[0])
  66 
  67 
  68 def reanalyze(db):
  69     events = [
  70         event_dict(row)
  71         for row in db.execute("SELECT * FROM events ORDER BY timestamp,id")
  72     ]
  73     alerts = detect(events, settings(db))
  74     statuses = dict(db.execute("SELECT id,status FROM alerts"))
  75     db.execute("DELETE FROM alerts")
  76     db.executemany(
  77         "INSERT INTO alerts VALUES (?,?,?)",
  78         [(a["id"], json.dumps(a), statuses.get(a["id"], "new")) for a in alerts],
  79     )
  80     return len(alerts)
  81 
  82 
  83 def ingest(db, events, filename):
  84     db.execute("BEGIN IMMEDIATE")
  85     before = db.execute("SELECT count(*) FROM events").fetchone()[0]
  86     inserted = 0
  87     for e in events:
  88         inserted += db.execute(
  89             "INSERT OR IGNORE INTO events VALUES (?,?,?,?,?,?,?,?,?,?)",
  90             (
  91                 e.id,
  92                 e.timestamp,
  93                 e.source,
  94                 e.event_type,
  95                 e.host,
  96                 e.user,
  97                 e.source_ip,
  98                 e.raw,
  99                 json.dumps(e.metadata),
 100                 filename,
 101             ),
 102         ).rowcount
 103     if before + inserted > MAX_EVENTS:
 104         raise ValueError(f"Local event limit is {MAX_EVENTS}; use a fresh database")
 105     count = (
 106         reanalyze(db)
 107         if inserted
 108         else db.execute("SELECT count(*) FROM alerts").fetchone()[0]
 109     )
 110     return {
 111         "inserted": inserted,
 112         "duplicates": len(events) - inserted,
 113         "alerts_total": count,
 114     }
