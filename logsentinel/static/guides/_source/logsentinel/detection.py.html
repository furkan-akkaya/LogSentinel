   3 from collections import defaultdict, deque
   4 from datetime import datetime
   5 import hashlib
   6 import json
   7 
   8 DEFAULTS = {
   9     "ssh_threshold": 5,
  10     "windows_threshold": 5,
  11     "sudo_threshold": 3,
  12     "success_threshold": 3,
  13     "window_seconds": 300,
  14 }
  15 RULES = {
  16     "ssh_brute_force": ("SSH password guessing", "high", ["T1110.001"]),
  17     "ssh_success_after_failures": (
  18         "SSH success after failures",
  19         "critical",
  20         ["T1110.001", "T1078"],
  21     ),
  22     "windows_brute_force": ("Windows repeated logon failures", "high", ["T1110.001"]),
  23     "windows_success_after_failures": (
  24         "Windows success after failures",
  25         "critical",
  26         ["T1110.001", "T1078"],
  27     ),
  28     "sudo_failures": ("Repeated sudo authentication failures", "medium", ["T1110.001"]),
  29 }
  30 THRESHOLD_SETTINGS = {
  31     "ssh_brute_force": "ssh_threshold",
  32     "windows_brute_force": "windows_threshold",
  33     "sudo_failures": "sudo_threshold",
  34     "ssh_success_after_failures": "success_threshold",
  35     "windows_success_after_failures": "success_threshold",
  36 }

  39 def seconds(stamp):
  40     return datetime.fromisoformat(stamp.replace("Z", "+00:00")).timestamp()

  43 def correlation_key(event):
  44     """Unknown identities never join; Windows domain and logon type are required."""
  45     if not event["host"] or not event["user"]:
  46         return None
  47     meta = event["metadata"]
  48     if event["event_type"] == "ssh_failure" and meta.get("method") == "publickey":
  49         # Trying several SSH keys is not evidence of password guessing.
  50         return None
  51     if event["event_type"] == "sudo_failure":
  52         # PAM detail and sudo summaries can describe the same attempts.
  53         # Ignore summaries rather than double counting.
  54         if meta.get("aggregate_summary") or not meta.get("tty"):
  55             return None
  56         return (event["host"], event["user"], meta.get("tty"))
  57     if not event["source_ip"]:
  58         return None
  59     if event["source"] == "windows":
  60         if not meta.get("domain") or not meta.get("logon_type"):
  61             return None
  62         return (
  63             event["host"].casefold(),
  64             event["source_ip"],
  65             event["user"].casefold(),
  66             meta["domain"].casefold(),
  67             meta["logon_type"],
  68         )
  69     return (event["host"], event["source_ip"], event["user"])

  72 def detect(events, settings):
  73     alerts = []
  74     window = settings["window_seconds"]
  75 
  76     def alert(rule, evidence, key, anchor):
  77         title, severity, mitre = RULES[rule]
  78         last = evidence[-1]
  79         # Stable ID for an episode while its evidence grows.
  80         identifier = hashlib.sha256(
  81             json.dumps([rule, key, anchor]).encode()
  82         ).hexdigest()
  83         count = sum(e["event_type"].endswith("failure") for e in evidence)
  84         reason = f"{count} matching authentication failure records within {window}s"
  85         if "success_after" in rule:
  86             reason += ", followed by a strictly later successful logon. Review for possible account compromise."
  87         else:
  88             reason += ". The configured threshold was reached; review the evidence."
  89         alerts.append(
  90             {
  91                 "id": identifier,
  92                 "rule": rule,
  93                 "title": title,
  94                 "severity": severity,
  95                 "mitre": mitre,
  96                 "first_seen": evidence[0]["timestamp"],
  97                 "last_seen": last["timestamp"],
  98                 "host": last["host"],
  99                 "source_ip": last["source_ip"],
 100                 "user": last["user"],
 101                 "reason": reason,
 102                 "parameters": {
 103                     "threshold": settings[THRESHOLD_SETTINGS[rule]],
 104                     "window_seconds": window,
 105                 },
 106                 "evidence": [e["id"] for e in evidence],
 107             }
 108         )
 109 
 110     for family, failure, success, threshold in [
 111         ("ssh", "ssh_failure", "ssh_success", settings["ssh_threshold"]),
 112         (
 113             "windows",
 114             "windows_failure",
 115             "windows_success",
 116             settings["windows_threshold"],
 117         ),
 118         ("sudo", "sudo_failure", None, settings["sudo_threshold"]),
 119     ]:
 120         groups = defaultdict(list)
 121         for event in events:
 122             if event["event_type"] in (failure, success):
 123                 key = correlation_key(event)
 124                 if key:
 125                     groups[key].append(event)
 126         for key, group in groups.items():
 127             group.sort(key=lambda e: (e["timestamp"], e["id"]))
 128             recent = deque()
 129             episode = []
 130             episode_alert = None
 131             for event in group:
 132                 t = seconds(event["timestamp"])
 133                 while recent and seconds(recent[0]["timestamp"]) < t - window:
 134                     recent.popleft()
 135                 if event["event_type"] == failure:
 136                     if not episode or t - seconds(episode[0]["timestamp"]) > window:
 137                         episode, episode_alert = [], None
 138                     episode.append(event)
 139                     recent.append(event)
 140                     if episode_alert is None and len(recent) >= threshold:
 141                         # Evidence spans a bounded window, even across old episode boundaries.
 142                         episode = list(recent)
 143                         rule = (
 144                             "sudo_failures"
 145                             if family == "sudo"
 146                             else f"{family}_brute_force"
 147                         )
 148                         alert(rule, episode, key, episode[0]["id"])
 149                         episode_alert = alerts[-1]
 150                     elif episode_alert is not None:
 151                         episode_alert["last_seen"] = event["timestamp"]
 152                         episode_alert["evidence"] = [e["id"] for e in episode]
 153                         episode_alert["reason"] = (
 154                             f"{len(episode)} matching authentication failure records within {window}s. The configured threshold was reached; review the evidence."
 155                         )
 156                 else:
 157                     # Equal-time records do not establish temporal precedence.
 158                     preceding = [
 159                         e for e in recent if e["timestamp"] < event["timestamp"]
 160                     ]
 161                     if len(preceding) >= settings["success_threshold"]:
 162                         alert(
 163                             f"{family}_success_after_failures",
 164                             preceding + [event],
 165                             key,
 166                             event["id"],
 167                         )
 168                     # Keep equal-time failures for a subsequent, strictly later
 169                     # success: their ordering relative to this success is unknown.
 170                     recent = deque(
 171                         e for e in recent if e["timestamp"] == event["timestamp"]
 172                     )
 173                     episode, episode_alert = [], None
 174     return sorted(alerts, key=lambda a: (a["last_seen"], a["id"]))
