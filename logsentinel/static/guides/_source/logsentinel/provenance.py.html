   1 """Identify exact normalized matches to bundled synthetic records, not realness."""
   2 
   3 from functools import lru_cache
   4 from pathlib import Path
   5 
   6 from .parsers import parse
   7 
   8 
   9 @lru_cache(maxsize=1)
  10 def demo_ids():
  11     root = Path(__file__).parent / "demo"
  12     return frozenset(
  13         event.id
  14         for name, fmt in [
  15             ("linux-auth.log", "linux"),
  16             ("windows-security.json", "json"),
  17         ]
  18         for event in parse((root / name).read_text(), fmt).events
  19     )
  20 
  21 
  22 def dataset_kind(identifier):
  23     return (
  24         "bundled_synthetic_match" if identifier in demo_ids() else "unverified_import"
  25     )
  26 
  27 
  28 def dataset_summary(events):
  29     matched = sum(e["id"] in demo_ids() for e in events)
  30     return {
  31         "bundled_synthetic_match": matched,
  32         "unverified_import": len(events) - matched,
  33     }
