  28 def validate_windows_source(system):
  29     # Absence retains the documented simplified JSON/XML compatibility.
  30     # Presence, including empty or malformed fields, must never imply trust.
  31     if "Channel" in system:
  32         channel = system["Channel"]
  33         if not isinstance(channel, str) or channel.strip().casefold() != "security":
  34             raise SourceMismatch("Windows source rejected: Channel must be Security")
  35     names, guids = [], []
  36     if "Provider" in system:
  37         provider = system["Provider"]
  38         if isinstance(provider, str):
  39             names.append(provider)
  40         elif isinstance(provider, dict):
  41             names.extend(provider[k] for k in ("Name", "@Name") if k in provider)
  42             guids.extend(provider[k] for k in ("Guid", "@Guid") if k in provider)
  43             if not names and not guids:
  44                 raise SourceMismatch(
  45                     "Windows source rejected: Provider needs Name or Guid"
  46                 )
  47         else:
  48             raise SourceMismatch("Windows source rejected: invalid Provider")
  49     if "ProviderName" in system:
  50         names.append(system["ProviderName"])
  51     if "ProviderGuid" in system:
  52         guids.append(system["ProviderGuid"])
  53     for name in names:
  54         if (
  55             not isinstance(name, str)
  56             or name.strip().casefold() != SECURITY_PROVIDER.casefold()
  57         ):
  58             raise SourceMismatch("Windows source rejected: unexpected provider name")
  59     for guid in guids:
  60         try:
  61             if (
  62                 not isinstance(guid, str)
  63                 or UUID(guid.strip()) != SECURITY_PROVIDER_GUID
  64             ):
  65                 raise ValueError
  66         except (ValueError, AttributeError):
  67             raise SourceMismatch(
  68                 "Windows source rejected: unexpected provider GUID"
  69             ) from None

  72 def timestamp(value: str, offset: str = "+00:00") -> str:
  73     if not re.fullmatch(r"[+-](?:0\d|1[0-4]):[0-5]\d", offset) or (
  74         offset[1:3] == "14" and offset[-2:] != "00"
  75     ):
  76         raise ValueError("Timezone must be an offset between -14:00 and +14:00")
  77     dt = datetime.fromisoformat(str(value).replace("Z", "+00:00"))
  78     if dt.tzinfo is None:
  79         minutes = int(offset[1:3]) * 60 + int(offset[4:])
  80         dt = dt.replace(
  81             tzinfo=timezone(
  82                 timedelta(minutes=minutes if offset[0] == "+" else -minutes)
  83             )
  84         )
  85     return (
  86         dt.astimezone(timezone.utc)
  87         .isoformat(timespec="microseconds")
  88         .replace("+00:00", "Z")
  89     )

  92 def ip(value):
  93     if not value or value in ("-", "unknown"):
  94         return None
  95     try:
  96         address = ipaddress.ip_address(str(value))
  97         if isinstance(address, ipaddress.IPv6Address) and address.ipv4_mapped:
  98             return str(address.ipv4_mapped)
  99         return str(address)
 100     except ValueError:
 101         return None

 125         # Raw formatting and upload filename are deliberately excluded.

 128             self.source,
 129             self.event_type,

 132             self.source_ip,

 154     r"^(?:<\d+>)?(?P<stamp>[A-Z][a-z]{2}\s+\d{1,2}\s+\d\d:\d\d:\d\d|\d{4}-\d\d-\d\dT\S+)\s+(?P<host>\S+)\s+(?P<process>sshd|sudo)(?:\[(?P<pid>\d+)\])?:\s*(?P<message>.*)$"

 161 def linux_event(line, year, offset):
 162     m = PREFIX.match(line)
 163     if not m:
 164         # Supported syslog but unrelated daemon, or a malformed record.
 165         if re.match(
 166             r"^(?:<\d+>)?(?:[A-Z][a-z]{2}\s+\d+\s+\d\d:\d\d:\d\d|\d{4}-\d\d-\d\dT\S+)\s+\S+\s+[^ :]+(?:\[\d+\])?:",
 167             line,
 168         ):
 169             return None
 170         raise ValueError("Unsupported Linux record or missing syslog prefix")
 171     stamp = m["stamp"]
 172     if not stamp[0].isdigit():
 173         stamp = datetime.strptime(f"{year} {stamp}", "%Y %b %d %H:%M:%S").isoformat()
 174     stamp = timestamp(stamp, offset)
 175     metadata = {"pid": m["pid"]}
 176     if m["process"] == "sshd":
 177         ssh = SSH.search(m["message"])
 178         if not ssh:
 179             return None
 180         metadata.update(method=ssh["method"], source_port=ssh["port"])
 181         source_ip = ip(ssh["ip"])
 182         if not source_ip:
 183             raise ValueError("Invalid SSH source IP")
 184         return Event(
 185             stamp,
 186             "linux",
 187             "ssh_success" if ssh["status"] == "Accepted" else "ssh_failure",
 188             m["host"],
 189             ssh["user"],
 190             source_ip,
 191             line,
 192             metadata,
 193         )
 194     message = m["message"]
 195     if "pam_unix(sudo:auth): authentication failure" in message:
 196         values = dict(re.findall(r"(\w+)=([^\s;]*)", message))
 197         user = clean(values.get("ruser")) or clean(values.get("logname"))
 198         metadata["target_user"] = clean(values.get("user"))
 199         metadata["tty"] = clean(values.get("tty"))
 200         return Event(
 201             stamp,
 202             "linux",
 203             "sudo_failure",
 204             m["host"],
 205             user,
 206             ip(values.get("rhost")),
 207             line,
 208             metadata,
 209         )
 210     bad = re.search(r"^\s*(\S+)\s*:\s*\d+ incorrect password attempts?", message)
 211     if bad:
 212         metadata["aggregate_summary"] = True
 213         return Event(
 214             stamp, "linux", "sudo_failure", m["host"], bad[1], None, line, metadata
 215         )
 216     return None

 223 def xml_record(node):
 224     system, data = {}, {}
 225     for child in node:
 226         if local_tag(child.tag) == "System":
 227             for item in child:
 228                 system[local_tag(item.tag)] = (
 229                     item.attrib
 230                     if local_tag(item.tag) in ("TimeCreated", "Provider")
 231                     else item.text
 232                 )
 233         if local_tag(child.tag) == "EventData":
 234             data = {item.attrib.get("Name", ""): item.text for item in child}
 235     return {"System": system, "EventData": data}

 238 def windows_event(record, raw, offset):
 239     if not isinstance(record, dict):
 240         raise ValueError("Windows record must be an object")
 241     record = record.get("Event", record)
 242     if not isinstance(record, dict):
 243         raise ValueError("Event must be an object")
 244     system = record.get("System", record)
 245     data = record.get("EventData", record)
 246     if not isinstance(system, dict) or not isinstance(data, dict):
 247         raise ValueError("System and EventData must be objects")
 248     if "Data" in data:
 249         items = data["Data"]
 250         if not isinstance(items, list):
 251             items = [items]
 252         data = {
 253             item.get("@Name", item.get("Name")): item.get("#text", item.get("Value"))
 254             for item in items
 255             if isinstance(item, dict)
 256         }
 257     event_id = system.get("EventID", system.get("Id"))
 258     if isinstance(event_id, dict):
 259         event_id = event_id.get("#text")
 260     if str(event_id) not in ("4624", "4625"):
 261         if event_id is None:
 262             raise ValueError("Missing EventID")
 263         return None
 264     validate_windows_source(system)
 265     stamp = system.get("TimeCreated", system.get("Timestamp"))
 266     if isinstance(stamp, dict):
 267         stamp = stamp.get("@SystemTime", stamp.get("SystemTime"))
 268     host = clean(system.get("Computer", system.get("MachineName")))
 269     if not stamp or not host:
 270         raise ValueError("Windows event needs TimeCreated and Computer")
 271     metadata = {
 272         "event_id": int(event_id),
 273         "record_id": clean(system.get("EventRecordID", system.get("RecordId"))),
 274         # Keep the existing event identity compatible with earlier imports.
 275         # Provider is checked against raw input, not added to identity metadata.
 276         "channel": "Security",
 277         "domain": clean(data.get("TargetDomainName")),
 278         "logon_type": clean(data.get("LogonType")),
 279     }
 280     return Event(
 281         timestamp(stamp, offset),
 282         "windows",
 283         "windows_success" if str(event_id) == "4624" else "windows_failure",
 284         host.lower(),
 285         clean(data.get("TargetUserName")),
 286         ip(data.get("IpAddress")),
 287         raw,
 288         metadata,
 289     )

 292 def parse(text, format="auto", year=2026, offset="+00:00"):
 293     timestamp("2000-01-01T00:00:00", offset)  # Validate even for empty files.
 294     if not 1970 <= year <= 2100:
 295         raise ValueError("Year must be between 1970 and 2100")
 296     text = text.lstrip("\ufeff").strip()
 297     if not text:
 298         raise ValueError("File is empty")
 299     if format == "auto":
 300         format = (
 301             "xml"
 302             if text.startswith("<") and not re.match(r"<\d+>", text)
 303             else "json"
 304             if text[0] in "[{"
 305             else "linux"
 306         )
 307     result = ParseResult()
 308     records = []
 309     if format == "xml":
 310         try:
 311             root = ET.fromstring(
 312                 text, forbid_dtd=True, forbid_entities=True, forbid_external=True
 313             )
 314         except (ET.ParseError, DefusedXmlException, RecursionError) as exc:
 315             raise ValueError("Invalid or unsafe XML document") from exc
 316         nodes = (
 317             [root]
 318             if local_tag(root.tag) == "Event"
 319             else [n for n in root if local_tag(n.tag) == "Event"]
 320         )
 321         if not nodes:
 322             raise ValueError("XML must contain Event or Events/Event")
 323         records = [(xml_record(n), ET.tostring(n, encoding="unicode")) for n in nodes]
 324     elif format == "json":
 325         try:
 326             doc = json.loads(text)
 327             if isinstance(doc, dict) and "Events" in doc:
 328                 doc = doc["Events"]
 329             docs = doc if isinstance(doc, list) else [doc]
 330             records = [(doc, json.dumps(doc, ensure_ascii=True)) for doc in docs]
 331         except (json.JSONDecodeError, RecursionError):
 332             records = [(line, line) for line in text.splitlines() if line.strip()]
 333     elif format == "linux":
 334         records = [(line, line) for line in text.splitlines() if line.strip()]
 335     else:
 336         raise ValueError("Format must be auto, linux, json or xml")
 337     if len(records) > MAX_RECORDS:
 338         raise ValueError(f"Import exceeds {MAX_RECORDS} records")
 339     for number, (record, raw) in enumerate(records, 1):
 340         try:
 341             if len(raw) > MAX_LINE:
 342                 raise ValueError("Record exceeds 64 KiB")
 343             if format == "linux":
 344                 event = linux_event(record, year, offset)
 345             else:
 346                 event = windows_event(
 347                     json.loads(record) if isinstance(record, str) else record,
 348                     raw,
 349                     offset,
 350                 )
 351             if event:
 352                 result.events.append(event)
 353             else:
 354                 result.ignored += 1
 355         except SourceMismatch as exc:
 356             result.reject(number, str(exc))
 357         except (ValueError, TypeError, AttributeError, OverflowError, RecursionError):
 358             result.reject(number, "Malformed record, invalid fields or timestamp")
 359     return result
